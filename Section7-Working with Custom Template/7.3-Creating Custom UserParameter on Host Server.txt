# Check the installed Python 3 version
python3 --version

# Find the path of the Python 3 executable
which python3

# Display the Python 3 version number only
python3 --version | awk '{print $2}'

# Open the custom Zabbix Agent 2 configuration file
vi /etc/zabbix/zabbix_agent2.d/luckydraw.conf

# Define a custom UserParameter to retrieve the Python version
UserParameter=python.version,/usr/bin/python3 --version | awk '{print $2}'

# Display the Zabbix Agent 2 main configuration file
cat /etc/zabbix/zabbix_agent2.conf

# Test the Zabbix Agent 2 configuration for errors
zabbix_agent2 -T -c /etc/zabbix/zabbix_agent2.conf

# Restart Zabbix Agent 2 to load the custom UserParameter
systemctl restart zabbix-agent2.service

# Test the custom Python version monitoring item
zabbix_agent2 -t python.version
