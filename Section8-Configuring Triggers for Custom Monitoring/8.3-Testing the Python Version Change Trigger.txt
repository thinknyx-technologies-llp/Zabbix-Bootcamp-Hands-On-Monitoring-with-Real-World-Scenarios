# Update the package lists
apt-get update

# Upgrade the installed packages
apt-get upgrade -y

# Check the currently installed Python 3 version
python3 --version

# Install the software-properties-common package
sudo apt install -y software-properties-common

# Add the Deadsnakes PPA to access additional Python versions
sudo add-apt-repository ppa:deadsnakes/ppa

# Update the package lists after adding the repository
sudo apt update

# Install Python 3.12, its virtual environment package, and development headers
sudo apt install -y python3.12 python3.12-venv python3.12-dev

# Verify the installed Python 3.12 version
python3.12 --version

# Check the default Python 3 version
python3 --version

# Register Python 3.10 as an alternative for the python3 command
sudo update-alternatives --install /usr/bin/python3 python3 /usr/bin/python3.10 1

# Register Python 3.12 as an alternative with a higher priority
sudo update-alternatives --install /usr/bin/python3 python3 /usr/bin/python3.12 2

# Select the default Python 3 version interactively
sudo update-alternatives --config python3

# Verify the selected Python 3 version
python3 --version

# Check the location of the Python executable
which python

# Check the location of the Python 3 executable
which python3
