# Identify the operating system and version
cat /etc/os-release

# Install Zabbix Agent based on the identified OS and version

# Open the Zabbix Agent configuration file
vi /etc/zabbix/zabbix_agentd.conf

# Configure the Zabbix Server's private IP address
# Server=<zabbix_private_ip>

# Restart the Zabbix Agent service
